// Ouverture f0 → f80 (2,7 s) : carte, point-personnage, aplat, chute et largeur de TYPE
const FPS = 30, BEAT = 60 / 128;            // 1 temps = 0,46875 s = 14,0625 im
const F = n => n / FPS;                      // images → secondes
const B = n => n * BEAT;                     // temps musicaux → secondes (grille dès 0)
const tl = gsap.timeline({ paused: true });  // #dot centré (xPercent/yPercent −50), left/top en %
// carte CV : expo.out 24 im dès t = 0 ; rotation plus courte (13 im, power2.out)
tl.from("#card", { yPercent: 153, duration: F(24), ease: "expo.out" }, 0)
  .from("#card", { rotation: -5.2, duration: F(13), ease: "power2.out" }, F(2))
  .from("#name .ch", { xPercent: 60, autoAlpha: 0, duration: F(4), ease: "power3.out", stagger: F(1) }, F(8))
  .from("#skills .row", { xPercent: 8, autoAlpha: 0, duration: F(5), ease: "power3.out", stagger: F(2) }, F(13))
// point final : pop back.out(3), pic +25 % à 6 im

// petit saut sur le temps 2 (1,9 % H)

// anticipation sur le demi-temps 2,5 : écrasé ×0,59 / ×1,25 en 6 im, tenu 4 im

  .fromTo("#dot", { scaleX: 0.5, scaleY: 1.3 }, { scaleX: 1, scaleY: 1, duration: F(6),
      ease: "power2.out", immediateRender: false }, B(3))
// gonflement plein cadre sur le demi-temps 3,5 : rayon ×57 en 5 im
