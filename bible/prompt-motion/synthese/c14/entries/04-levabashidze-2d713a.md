const FPS = 60, BEAT = 60 / 128;                 // 1 temps = 28,125 im
const F = n => n / FPS;                          // images → secondes
const B = n => n * BEAT;                         // temps musical → secondes
// ressort mesuré sur « LINEAR. » : ζ 0,34, 2,35 Hz (période ≈ 1 temps, dépassement 32 %)
const spring = (z = 0.34, hz = 2.35, dur = 0.75) => p => {
  const w = 2 * Math.PI * hz, t = p * dur, wd = w * Math.sqrt(1 - z * z);
  return 1 - Math.exp(-z * w * t) * (Math.cos(wd * t) + (z * w / wd) * Math.sin(wd * t));
};
const tl = gsap.timeline({ paused: true });
// temps 4 (f112,5) : la balle a rempli le cadre ; « EASE IN. » en expo.in, 14 im (arrêt au demi-temps 4,5)
tl.fromTo("#easeIn", { x: -470 }, { x: 0, duration: F(14), ease: "expo.in" }, B(4))
// temps 5 : « EASE OUT. » arrive de hors cadre droit, expo.out 22 im
