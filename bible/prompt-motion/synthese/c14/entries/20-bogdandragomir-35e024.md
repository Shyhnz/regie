const FPS = 60, BEAT = 60 / 96;              // 1 temps = 0,625 s = 37,5 im
const F = n => n / FPS;                      // images → secondes
const B = n => n * BEAT;                     // temps musicaux → secondes (grille dès 0 s)
const tl = gsap.timeline({ paused: true });
// temps 8 = 1er temps de la mesure 3 : le titre se re-décode gauche → droite
tl.to("#title1 .ch", { opacity: 0, duration: F(3), stagger: F(0.5) }, B(8) + F(1))
  .from("#title2 .ch", { opacity: 0, duration: F(5), ease: "power1.inOut",
      stagger: F(0.5) }, B(8) + F(4));
// 4 traînées : arrivée 0,5 im avant chaque demi-temps (f337, f355, f374, f393)
["#r3", "#r4", "#r5", "#r6"].forEach((r, i) => {
  const land = B(9 + i * 0.5) - F(0.5);
  tl.fromTo(r + " .streak", { left: "86%" }, { left: "47.2%", duration: F(9),
      ease: "sine.out" }, land - F(9))          // départ ¼ de temps avant

// 59 im fixes, puis temps 12 = mesure 4 : volet tramé lancé, il sort encore en mouvement
tl.fromTo("#wipe", { left: "-3%" }, { left: "113%", duration: F(27),
    ease: "power1.out" }, B(12) + F(1));          // bord visible f451 → f470
// petite liseuse : exactement 1 temps, de demi-temps à demi-temps

    ease: "power4.out" }, B(17.5))
  .set("#reader2 .ink", { autoAlpha: 1 }, B(18.5))              // flash noir 6 im
  .fromTo("#reader2 .page", { clipPath: "inset(0 0 100% 0)" },
    { clipPath: "inset(0 0 0% 0)", duration: F(7), ease: "power1.inOut" }, B(18.5) + F(6));
// temps 20 = mesure 6 : écran noir 6 im, puis iris linéaire jusqu'au coin du cadre

  .fromTo("#iris", { scale: 0 }, { scale: 1, duration: F(20), ease: "none" }, B(20) + F(7))
