**Leçons**

1. À 128 BPM et 60 fps, temps n = round(n × 28,125) im : couper exactement sur cette image ; les barres détectées dans l'audio tombent 1 im plus tard.
2. Faire finir le mouvement sur le temps : l'en-tête arrive 1 im avant la coupe, coup de zoom et sortie accélèrent (power2-4.in) jusqu'à l'image de coupe.
3. Saut d'objet d'un temps : montée power1.out 13 im, chute power1.in 15 im, x linéaire, étiré 1,6:1 au départ, écrasé 1,9:1 à l'impact, rond en 4 im.
4. Décalages courts et réguliers : 1 im par lettre qui monte, 2 im par lettre qui tombe ou par ligne empilée, 7 im (1/4 de temps) par ligne de phrase.
5. Rafale avant le drop : 4 plans de 7 im, 6 plans de 3-4 im, puis 14 im de noir avec un seul point ; drop sur le 1er temps de mesure.
6. Après le drop, un accent tous les 1,5 temps (42 im) ; le point atterrit 14 im après chaque coupe ; gel de 22-24 im pour lire la chute.

// Ouverture f0 → f197 (3,3 s) : rebond, saut-raccord, phrase en 1/16, en-tête, SQUASH
const FPS = 60, BEAT = 60 / 128;            // 1 temps = 0,46875 s = 28,125 im
const F = n => n / FPS;                      // images → secondes
const B = n => n * BEAT;                     // temps musicaux → secondes
const tl = gsap.timeline({ paused: true });  // #dot centré (xPercent/yPercent -50), left/top en %
// petit rebond d'un temps : élan écrasé f0-f2, sommet au demi-temps, retour sur le temps 1
tl.to("#dot", { scaleX: 1.15, scaleY: 0.87, duration: F(2), ease: "none" }, 0)
  .to("#dot", { top: "46.7%", scaleX: 1, scaleY: 1, duration: F(13), ease: "power1.out" }, F(2))
  .to("#dot", { top: "69.1%", duration: F(13), ease: "power1.in" }, F(15))
// grand saut = 1 temps exact : x linéaire, y en parabole, étiré 1,6:1 au départ
  .to("#dot", { left: "50.1%", duration: B(1), ease: "none" }, B(1))
  .fromTo("#dot", { scaleX: 0.8, scaleY: 1.27 }, { top: "29.4%", scaleX: 1, scaleY: 1,
      duration: F(13), ease: "power1.out", immediateRender: false }, B(1))
  .to("#dot", { top: "85.5%", scale: 0.72, duration: F(15), ease: "power1.in" }, B(1) + F(13))
// coupe sur le temps 2 : impact écrasé 1,9:1, rond en 4 im, OPUS monte lettre à lettre
  .set("#hi", { autoAlpha: 0 }, B(2)).set("#opus", { autoAlpha: 1 }, B(2))
  .fromTo("#dot", { scaleX: 0.98, scaleY: 0.52 }, { scaleX: 0.72, scaleY: 0.72,
      duration: F(4), ease: "power2.out", immediateRender: false }, B(2) + F(1))
  .from("#opus .ch", { yPercent: 110, duration: F(8), ease: "power3.out", stagger: F(1) }, B(2) + F(1))
// coupe sur le temps 4 : 4 lignes décalées d'1/4 de temps (7 im), lettres à 1 im
  .set("#opus", { autoAlpha: 0 }, B(4)).set("#watch", { autoAlpha: 1 }, B(4))
  .from("#watch .line", { yPercent: 105, duration: F(12), ease: "power3.out", stagger: B(0.25) }, B(4) + F(1))
// le point re-saute (1 temps) et devient le point de DO. sur le temps 5
  .to("#dot", { left: "87.3%", duration: B(1), ease: "none" }, B(4))
  .to("#dot", { top: "43.4%", duration: F(14), ease: "power1.out" }, B(4))

// « I CAN » devient l'en-tête : 10 im, ARRIVE 1 im avant la coupe du temps 6
  .to("#ican", { top: "6.1%", scale: 0.56, transformOrigin: "0% 0%",
      duration: F(10), ease: "power2.inOut" }, B(6) - F(11))
  .set("#watch", { autoAlpha: 0 }, B(6)).set("#list", { autoAlpha: 1 }, B(6))
// SQUASH : chute 4 im (sine.in), 2 im entre lettres, ressort 0,49 → 1,17 → 0,93 → 1
